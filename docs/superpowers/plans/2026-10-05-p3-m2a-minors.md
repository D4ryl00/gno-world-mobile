## Review Focus

- **The app comes back after a long background in which more than 200 lines arrived:** the loop resumes from its cursor and keeps polling until it holds the newest line (Task 2, "resume catches up past one page").
- **The log was reset (new gen) while the app was in the background:** on resume the view reloads and the app's "mine", reactions and votes are forgotten (Task 2, "a resume that meets a new gen reloads and forgets").
- **The name sheet is opened by Send, then closed without a name:** the held-back line is dropped, not sent later when a name is saved from the header (Task 7, "closing the sheet without a name drops the held-back line").
- **More than 400 own reactions are remembered:** the oldest lines' marks are dropped, as the site's `br.rx` keeps 400 (Task 4, "the remembered marks keep the newest 400 lines").
- **An edit whose text differs only by surrounding spaces:** nothing is sent and the line is not marked edited (Task 5, "an edit that changes nothing sends nothing").

`store.ts`:

```ts
export const MARKS_KEY = 'br.chat.marks';
/** As the site's br.rx: the newest 400 lines' marks are kept. */
const MARKS_MAX = 400;

const newest = <T>(m: Record<number, T>): Record<number, T> =>
```

```ts
import { ImageManipulator, SaveFormat } from 'expo-image-manipulator';

/** Avatars show at a few dozen pixels: a camera photo is cut down before it is sent. */
export const AVATAR_PX = 256;

/** The picked photo as a JPEG AVATAR_PX wide, in bytes (read from base64: no Response.blob() round trip). */
export async function avatarJpeg(uri: string): Promise<Uint8Array> {
  const image = await ImageManipulator.manipulate(uri).resize({ width: AVATAR_PX }).renderAsync();
  const out = await image.saveAsync({ format: SaveFormat.JPEG, compress: 0.8, base64: true });
  if (!out.base64) throw new Error('the picture came back empty');
  return Uint8Array.from(atob(out.base64), (c) => c.charCodeAt(0));
}
```

- [ ] **Step 3: Implement**

`src/hooks/use-keyboard-height.ts`:

```ts
import { useEffect, useState } from 'react';
import { Keyboard } from 'react-native';

/** The software keyboard's height while it is up, 0 otherwise. */
export function useKeyboardHeight(): number {
  const [height, setHeight] = useState(0);
  useEffect(() => {
    const show = Keyboard.addListener('keyboardDidShow', (e) => setHeight(e.endCoordinates.height));
    const hide = Keyboard.addListener('keyboardDidHide', () => setHeight(0));
    return () => {
      show.remove();
      hide.remove();
    };
  }, []);
  return height;
}
```

`src/components/Page.tsx`:

```tsx
import type { PropsWithChildren } from 'react';

import { useKeyboardHeight } from '@/hooks/use-keyboard-height';

/**
 * A screen's scrolling body when it has fields. With the keyboard up, a tap on a button acts at once (instead of
 * only closing the keyboard), the content can scroll above the keyboard (iOS: the scroll view's insets; Android,
 * edge-to-edge: the window keeps its size, so room the keyboard's height is added under the content), and dragging
 * closes the keyboard.
 */
export function Page({ style, children }: PropsWithChildren<{ style: StyleProp<ViewStyle> }>) {
  const keyboard = useKeyboardHeight();
  return (
    <ScrollView
      contentContainerStyle={style}
      keyboardShouldPersistTaps="handled"
      keyboardDismissMode="on-drag"
      automaticallyAdjustKeyboardInsets>
      {children}
      {Platform.OS === 'android' && keyboard > 0 ? <View testID="keyboard-room" style={{ height: keyboard }} /> : null}
    </ScrollView>
  );
}
```
